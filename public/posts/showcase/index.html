BLOCKSIZE = 8

def blocks(x):
    return list(zip(*[iter(x)] * BLOCKSIZE))

def permute(s):
    res = [0 for _ in s]
    for b in range(8):
        for i, x in enumerate(s):
            res[PBOXES[b][i]] |= (x & (1 << b))
    return res

def H(m):
    assert len(m) % BLOCKSIZE == 0
    state = [0 for _ in range(18)]
    for b in blocks(m):
        for i, c in enumerate(b):
            state[i] ^= c
        for _ in range(8):
            state = permute(state)
            state = [SBOX[b] for b in state]
    return bytes(state)
